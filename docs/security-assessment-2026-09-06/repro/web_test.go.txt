func assessmentCookie(t *testing.T, s *Server) (*http.Cookie, uint32) {
	t.Helper()
	u, err := s.id.Store().CreateUser(context.Background(), "AUDITADMIN", "audit@example.invalid", "!migrated", 3)
	if err != nil {
		t.Fatal(err)
	}
	rec := httptest.NewRecorder()
	sess := s.sessions.GetOrCreate(rec, httptest.NewRequest("GET", "/", nil))
	sess.Update(func(d *session.Data){d.User=toSiteUser(u)})
	s.sessions.SaveLatest(sess)
	return rec.Result().Cookies()[0], u.ID
}
func TestAssessmentKBRetainsDemotedPrivilege(t *testing.T) {
	ts, s := testWeb(t)
	defer ts.Close()
	cookie, uid := assessmentCookie(t, s)
	if err := s.id.Store().SetStaffLevel(context.Background(), uid, 2); err != nil {
		t.Fatal(err)
	}
	req := httptest.NewRequest("GET", "/kb/admin", nil)
	req.AddCookie(cookie)
	w := httptest.NewRecorder()
 s.Handler().ServeHTTP(w,req)
 if w.Code!=http.StatusUnauthorized && w.Code!=http.StatusForbidden {t.Fatalf("demoted session authorized: %d",w.Code)}
 t.Log("Fixed: KB rejects demoted session")
}
func TestAssessmentMultipartLeaksEarlierFiles(t *testing.T) {
	ts, s := testWeb(t)
	defer ts.Close()
	cookie, _ := assessmentCookie(t, s)
	var body bytes.Buffer
	mw := multipart.NewWriter(&body)
	for _, name := range []string{"one.zip", "two.zip"} {
		p, err := mw.CreateFormFile("file", name)
		if err != nil {
			t.Fatal(err)
		}
		p.Write([]byte("synthetic"))
	}
	mw.Close()
	req := httptest.NewRequest("POST", "/staff/downloads/upload", &body)
	req.Header.Set("Content-Type", mw.FormDataContentType())
	req.AddCookie(cookie)
	s.staffDownloadPOST(httptest.NewRecorder(), req)
	entries, err := os.ReadDir(s.cfg.DownloadsDir)
	if err != nil {
		t.Fatal(err)
	}
	n := 0
	for _, e := range entries {
		if strings.HasPrefix(e.Name(), ".up-") {
			n++
		}
	}
	if n != 1 {
		t.Fatalf("expected 1 leaked staging file, got %d", n)
	}
	t.Log("Confirmed: rejected two-file upload leaves first staged file on disk")
}
func TestAssessmentResetTokenLogged(t *testing.T) {
	ts, s := testWeb(t)
	defer ts.Close()
	var logs bytes.Buffer
	s.log = slog.New(slog.NewTextHandler(&logs, nil))
	const token = "SYNTHETIC-RESET-TOKEN-NOT-A-SECRET"
	req := httptest.NewRequest("GET", "/account/reset/"+token, nil)
	s.Handler().ServeHTTP(httptest.NewRecorder(), req)
	if !strings.Contains(logs.String(), token) {
		t.Fatal("token path not logged")
	}
	t.Log("Confirmed: middleware includes reset-token path in access log")
}
func TestAssessmentPendingMFACompletesAfterBan(t *testing.T) {
	ts, s := testWeb(t)
	defer ts.Close()
	ctx := context.Background()
	u, err := s.id.Register(ctx, "AuditPlayer", "AuditPass123", "audit@example.invalid", "", "AuditPass123", 2)
	if err != nil {
		t.Fatal(err)
	}
	secret, _, err := s.id.Store().StartTOTP(ctx, u.ID, u.Username, "Audit")
	if err != nil {
		t.Fatal(err)
	}
	code, err := totp.GenerateCode(secret, time.Now())
	if err != nil {
		t.Fatal(err)
	}
	if _, err = s.id.Store().ConfirmTOTP(ctx, u.ID, code); err != nil {
		t.Fatal(err)
	}
	rec := httptest.NewRecorder()
	sess := s.sessions.GetOrCreate(rec, httptest.NewRequest("GET", "/", nil))
	u,err=s.id.GetByID(ctx,u.ID);if err!=nil{t.Fatal(err)}
 sess.Update(func(d *session.Data){d.PendingUser=toSiteUser(u);d.PendingExpiry=time.Now().Add(time.Minute)})
 csrf:=sess.Snapshot().CSRF
	s.sessions.SaveLatest(sess)
	if err := s.accounts.Ban(ctx, 4, "AUDITOR", u.Username, "permanent", "fixture suspension"); err != nil {
		t.Fatal(err)
	}
	s.sessions.RevokeUser(u.ID, "")
	form := url.Values{"csrf_token": {csrf}, "code": {code}}
	req := httptest.NewRequest("POST", "/account/login/totp", strings.NewReader(form.Encode()))
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	req.AddCookie(rec.Result().Cookies()[0])
	w := httptest.NewRecorder()
	s.Handler().ServeHTTP(w, req)
	next := httptest.NewRequest("GET", "/downloads", nil)
	for _, c := range w.Result().Cookies() {
		next.AddCookie(c)
	}
	if got := s.sessions.GetOrCreate(httptest.NewRecorder(), next); got.Snapshot().User != nil {
		t.Fatal("banned pending user authenticated")
	}
	t.Log("Fixed: pending MFA cannot complete after ban and revocation")
}
